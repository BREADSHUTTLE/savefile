ONESTORE_MANAGER_BACKUP="${WORKSPACE}/OneStoreIAPManager.cs.backup"
ONESTORE_MANAGER_META_BACKUP="${WORKSPACE}/OneStoreIAPManager.cs.meta.backup"

exclude_onestore() {
  echo "=== Excluding OneStore files for iOS build ==="
  
  if [ -d "$ONESTORE_DIR" ]; then
    mv "$ONESTORE_DIR" "$ONESTORE_BACKUP"
    mv "$ONESTORE_META" "$ONESTORE_META_BACKUP" 2>/dev/null || true
    echo "✅ OneStoreCorpPlugins 폴더 임시 제외"
  fi
  
  if [ -f "$ONESTORE_MANAGER" ]; then
    mv "$ONESTORE_MANAGER" "$ONESTORE_MANAGER_BACKUP"
    mv "$ONESTORE_MANAGER_META" "$ONESTORE_MANAGER_META_BACKUP" 2>/dev/null || true
    echo "✅ OneStoreIAPManager.cs 임시 제외"
  fi
}

restore_onestore() {
  echo "=== Restoring OneStore files ==="
  
  if [ -d "$ONESTORE_BACKUP" ]; then
    mv "$ONESTORE_BACKUP" "$ONESTORE_DIR"
    mv "$ONESTORE_META_BACKUP" "$ONESTORE_META" 2>/dev/null || true
    echo "✅ OneStoreCorpPlugins 폴더 복원"
  fi
  
  if [ -f "$ONESTORE_MANAGER_BACKUP" ]; then
    mv "$ONESTORE_MANAGER_BACKUP" "$ONESTORE_MANAGER"
    mv "$ONESTORE_MANAGER_META_BACKUP" "$ONESTORE_MANAGER_META" 2>/dev/null || true
    echo "✅ OneStoreIAPManager.cs 복원"
  fi
}

# 로그 업로드 함수 (스레드에 첨부)
upload_log_to_slack() {
  local THREAD_TS=$1
  echo "=== Uploading Build Log to Slack ==="
  LOG_FILE="${WORKSPACE}/build.log"
  LOG_SIZE=$(stat -f%z "$LOG_FILE")
  
  RESPONSE=$(curl -s -X POST \
    -H "Authorization: Bearer ${SLACK_TOKEN}" \
    -H "Content-Type: application/x-www-form-urlencoded; charset=utf-8" \
    --data "filename=FAILED_Dev_iOS_log_${BUILD_NUMBER}.txt&length=${LOG_SIZE}" \
    https://slack.com/api/files.getUploadURLExternal)
  
  UPLOAD_URL=$(echo $RESPONSE | python3 -c "import sys,json; d=json.load(sys.stdin); print(d.get('upload_url',''))")
  FILE_ID=$(echo $RESPONSE | python3 -c "import sys,json; d=json.load(sys.stdin); print(d.get('file_id',''))")
  
  curl -s -X POST "$UPLOAD_URL" -F file=@"$LOG_FILE"
  
  if [ -n "$THREAD_TS" ]; then
    THREAD_JSON=",\"thread_ts\":\"${THREAD_TS}\""
  else
    THREAD_JSON=""
  fi
  
  curl -s -X POST \
    -H "Authorization: Bearer ${SLACK_TOKEN}" \
    -H "Content-Type: application/json; charset=utf-8" \
    --data "{\"files\":[{\"id\":\"${FILE_ID}\",\"title\":\"Dev iOS Build Log #${BUILD_NUMBER} (FAILED)\"}],\"channel_id\":\"C0A12TDUBK3\"${THREAD_JSON}}" \
    https://slack.com/api/files.completeUploadExternal
}

# 변경사항을 스레드로 전송하는 공통 함수
send_changes_thread() {
  local THREAD_CHANNEL=$1
  local THREAD_TS=$2
  local THREAD_CHANGES=$3
  
  if [ -n "$THREAD_TS" ] && [ -n "$THREAD_CHANGES" ]; then
    python3 << PYEOF
import json
import urllib.request

channel = "${THREAD_CHANNEL}"
thread_ts = "${THREAD_TS}"
token = "${SLACK_TOKEN}"
changes = '''${THREAD_CHANGES}'''

if changes.strip():
    payload = json.dumps({
        "channel": channel,
        "thread_ts": thread_ts,
        "text": f"📝 *변경사항:*\n{changes}"
    })
    req = urllib.request.Request(
        "https://slack.com/api/chat.postMessage",
        data=payload.encode('utf-8'),
        headers={
            "Authorization": f"Bearer {token}",
            "Content-Type": "application/json; charset=utf-8"
        }
    )
    resp = urllib.request.urlopen(req)
    resp_data = json.loads(resp.read().decode('utf-8'))
    print(f"스레드 전송 결과: {resp_data.get('ok')}")
PYEOF
  fi
}

# 실패 메시지 전송 함수

  restore_onestore

token = "${SLACK_TOKEN}"
build_num = "${BUILD_NUMBER}"
stage = "${FAIL_STAGE}"
build_type_msg = "${BUILD_TYPE_MSG}"

full_version = f"{app_version}.{build_num}"

skip_identity_warning = "\n⚠️ 본인인증 스킵 모드" if skip_identity == "true" else ""

message = f"❌ *ATOZPOKER Dev iOS Build #{build_num}* 실패했어요! ({stage}){skip_identity_warning}\n\n📦 *빌드 타입:* {build_type_msg}\n📱 *버전:* {full_version}\n🗂️ *리소스:* {resource_mode_msg}"

payload = json.dumps({
    "channel": channel,
    "text": message
})

req = urllib.request.Request(
    "https://slack.com/api/chat.postMessage",
    data=payload.encode('utf-8'),
    headers={
        "Authorization": f"Bearer {token}",
        "Content-Type": "application/json; charset=utf-8"
    }
)
response = urllib.request.urlopen(req)
resp_data = json.loads(response.read().decode('utf-8'))
print(resp_data.get("ts", ""), end="")
PYEOF
  )
  
  echo "실패 메인 메시지 ts: $MAIN_TS"
  send_changes_thread "C0A12TDUBK3" "$MAIN_TS" "$PLASTIC_CHANGES"
}

# Plastic SCM 변경사항 가져오기
get_plastic_changes() {
  LAST_BUILD_FILE="${WORKSPACE}/.last_build_changeset_dev_ios"
  
  CURRENT_BRANCH=$(cd "${WORKSPACE}" && $CM_PATH status --header 2>/dev/null | grep -o '/[^@]*' | head -1)
  CURRENT_CS=$(cd "${WORKSPACE}" && $CM_PATH status --header 2>/dev/null | grep -o 'cs:[0-9]*' | sed 's/cs://')
  
  if [ -f "$LAST_BUILD_FILE" ]; then
    LAST_CS=$(cat "$LAST_BUILD_FILE")
  else
    LAST_CS=0
  fi
  
  CHANGES=$(cd "${WORKSPACE}" && $CM_PATH log --csformat="{changesetid}|{branch}|✨ {comment} ({owner})" 2>/dev/null | awk -F'|' -v br="$CURRENT_BRANCH" -v last="$LAST_CS" -v curr="$CURRENT_CS" '$2 == br && $1 > last && $1 <= curr {print $3}')
  
  CHANGES=$(echo "$CHANGES" | sed 's/✨  (/✨ 코멘트가 없어요! (/g')
  
  if [ -z "$CHANGES" ]; then
    CHANGES="✨ 이전 빌드랑 변경된 내용이 없어요!"
  fi
  
  echo "$CHANGES"
}

save_current_changeset() {
  CURRENT_CS=$(cd "${WORKSPACE}" && $CM_PATH status --header 2>/dev/null | grep -o 'cs:[0-9]*' | sed 's/cs://')
  if [ -n "$CURRENT_CS" ]; then
    echo "$CURRENT_CS" > "${WORKSPACE}/.last_build_changeset_dev_ios"
    echo "Saved current changeset: ${CURRENT_CS}"
  fi
}
